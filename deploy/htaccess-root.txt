# Place this file as /elattar/.htaccess on the server (document root).
# The deploy workflow copies it automatically on each release.

Options -Indexes -MultiViews

<IfModule mod_rewrite.c>
    RewriteEngine On

    # Force HTTPS
    RewriteCond %{HTTPS} !=on
    RewriteCond %{HTTP:X-Forwarded-Proto} !https
    RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

    # Preserve Authorization header (API / Bearer tokens behind Apache)
    RewriteCond %{HTTP:Authorization} .
    RewriteRule ^ - [E=HTTP_AUTHORIZATION:%{HTTP:Authorization}]

    # Allow ACME / SSL validation
    RewriteRule ^\.well-known/ - [L]

    # Block direct web access to sensitive paths at document root
    RewriteRule ^(?:\.env|\.env\.|storage|vendor|bootstrap/cache|node_modules|tests)(?:/|$) - [F,L,NC]

    # If the request is not already under /public, route it there
    RewriteCond %{REQUEST_URI} !^/public/
    RewriteRule ^(.*)$ public/$1 [L]
</IfModule>

# Deny hidden files (.env, .git, etc.)
<FilesMatch "(^|/)\.">
    Require all denied
</FilesMatch>

# Explicitly block .env even if rewrite is disabled
<Files ".env">
    Require all denied
</Files>

<FilesMatch "^\.env\.">
    Require all denied
</FilesMatch>
